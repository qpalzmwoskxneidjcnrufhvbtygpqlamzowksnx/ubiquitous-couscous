using Unity.Notifications.Android;

// 1. Register Android Channel
var channel = new AndroidNotificationChannel()
{
    Id = "energy_channel",
    Name = "Game Reminders",
    Importance = Importance.Default,
    Description = "Reminds you when game events finish",
};
AndroidNotificationCenter.RegisterNotificationChannel(channel);

// 2. Schedule Notification
var notification = new AndroidNotification();
notification.Title = "Energy Refilled!";
notification.Text = "Jump back in and start playing!";
notification.FireTime = System.DateTime.Now.AddHours(2);

AndroidNotificationCenter.SendNotification(notification, "energy_channel");
